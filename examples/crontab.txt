# Add with: crontab -e   (set both absolute paths; cron uses the server's local time)
# flock keeps two runs of the same job from overlapping.
SA=/home/you/postweft
POSTWEFT_HOME=/home/you/.config/postweft
# Create the logs/ directory in POSTWEFT_HOME before adding these jobs.

# publish due posts
*/5 * * * *   cd "$SA" && flock -n /tmp/sa-publish.lock node bin/publish-queue.mjs >> "$POSTWEFT_HOME/logs/cron.log" 2>&1

# answer comments on your own posts (08:00-23:59)
*/20 8-23 * * * cd "$SA" && flock -n /tmp/sa-replies.lock node bin/reply-comments.mjs >> "$POSTWEFT_HOME/logs/cron.log" 2>&1

# answer Instagram DMs (08:00-23:59); waits for the comments job instead of skipping
*/15 8-23 * * * cd "$SA" && flock -w 600 /tmp/sa-replies.lock node bin/reply-dms.mjs >> "$POSTWEFT_HOME/logs/cron.log" 2>&1

# keep the Threads token alive (it lasts 60 days; renewed every Monday)
30 9 * * 1    cd "$SA" && node bin/refresh-tokens.mjs >> "$POSTWEFT_HOME/logs/cron.log" 2>&1

# evening summary to your phone
0 22 * * *    cd "$SA" && node bin/summary.mjs >> "$POSTWEFT_HOME/logs/cron.log" 2>&1
